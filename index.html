let desafios = []; // O array inicia vazio e puxará do arquivo externo let desafioAtualIndex = 0; let pontuacao = 0; // Referências dos Elementos DOM const inputLargura = document.getElementById('inputLargura'); const inputAltura = document.getElementById('inputAltura'); const inputProfundidade = document.getElementById('inputProfundidade'); const textVolume = document.getElementById('textVolume'); const motor = document.getElementById('custom-motor-geometrico'); const textEnunciado = document.getElementById('textEnunciado'); const inputRespostaAluno = document.getElementById('inputRespostaAluno'); const btnVerificar = document.getElementById('btnVerificar'); const textPontuacao = document.getElementById('textPontuacao'); const boxFeedback = document.getElementById('boxFeedback'); const interactiveControls = document.getElementById('interactive-controls'); // Fluxo Inicial da Aplicação document.addEventListener('DOMContentLoaded', async () => { atualizarDimensoes(); // CARREGA AS QUESTÕES DINAMICAMENTE DO ARQUIVO JSON EXTERNO try { // Adiciona um timestamp (?t=...) para quebrar o cache agressivo do navegador const respostaJSON = await fetch(`questoes.json?t=${new Date().getTime()}`); if (!respostaJSON.ok) throw new Error(); desafios = await respostaJSON.json(); carregarDesafio(); } catch (err) { textEnunciado.innerHTML = "

Erro ao carregar o banco de desafios geométricos.

"; } // Configuração dos Eventos dos Inputs do Motor inputLargura.addEventListener('input', atualizarDimensoes); inputAltura.addEventListener('input', atualizarDimensoes); inputProfundidade.addEventListener('input', atualizarDimensoes); // Evento do Botão de Validação btnVerificar.addEventListener('click', verificarResposta); }); function atualizarDimensoes() { const larg = inputLargura.value || "160"; const alt = inputAltura.value || "100"; const prof = inputProfundidade.value || "80"; const volume = Number(larg) * Number(alt) * Number(prof); textVolume.innerText = `Volume Total: ${volume.toLocaleString()} u³.`; if (motor) { motor.setAttribute('largura', larg); motor.setAttribute('altura', alt); motor.setAttribute('profundidade', prof); } } function carregarDesafio() { boxFeedback.style.display = 'none'; if (desafios.length > 0 && desafioAtualIndex < desafios.length) { textEnunciado.innerHTML = desafios[desafioAtualIndex].enunciado; inputRespostaAluno.value = ""; } else if (desafios.length > 0) { textEnunciado.innerHTML = "

🏆 Incrível, você completou todos os desafios do GeoMatrix!

"; interactiveControls.style.display = 'none'; textPontuacao.style.backgroundColor = '#D4EFDF'; textPontuacao.style.color = '#196F3D'; textPontuacao.style.borderColor = '#2ECC71'; } } function verificarResposta() { if (desafios.length === 0) return; const respostaDigitada = Number(inputRespostaAluno.value); const respostaCerta = Number(desafios[desafioAtualIndex].resposta_correta); if (respostaDigitada === respostaCerta) { pontuacao += 10; textPontuacao.innerText = `Pontos: ${pontuacao}`; boxFeedback.className = "feedback-box feedback-success"; boxFeedback.innerHTML = "✅ Resposta correta! Excelente raciocínio geométrico!"; boxFeedback.style.display = 'block'; setTimeout(() => { desafioAtualIndex++; carregarDesafio(); }, 3000); } else { boxFeedback.className = "feedback-box feedback-error"; boxFeedback.innerHTML = "❌ Resposta incorreta. Dica: Use o simulador 3D ao lado colocando as dimensões do enunciado para visualizar o sólido e conferir seus cálculos!"; boxFeedback.style.display = 'block'; } }
